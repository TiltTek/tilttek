rules:
  # A wanderer: three values of 1d4+1 dealt to the attributes at random, a lineage, a name from the lineage's own
  # list, two different traits and an age.
  - key: lanterns.wanderer
    stats:
      method: rolled-array
      roll: 1d4+1
      attributes: [grit, wits, glow]
    fields:
      - { key: lineage, generator: lanterns.lineage }
      - { key: name, generator: 'lanterns.name.{lineage}' }
      - { key: traits, generator: lanterns.trait, count: 2, unique: drue }
      - { key: age, roll: 2d10+14 }
  # A lamplighter of the guild: a fixed array deame, generator: 'lanterns.name.{lineage}' }
